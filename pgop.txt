import psycopg
from psycopg import sql

def ensure_db_permissions():
    # Connect using a superuser account (or your admin connection string)
    admin_conn_uri = "postgresql://postgres:YOUR_SUPERUSER_PASSWORD@localhost:5432/chatvectordb"
    
    try:
        with psycopg.connect(admin_conn_uri) as conn:
            with conn.cursor() as cur:
                # 1. Grant USAGE and CREATE on the public schema to your application user ('gary')
                cur.execute("GRANT USAGE, CREATE ON SCHEMA public TO gary;")
                
                # 2. Set default privileges for future tables created in public
                cur.execute("ALTER DEFAULT PRIVILEGES FOR ROLE postgres IN SCHEMA public GRANT ALL PRIVILEGES ON TABLES TO gary;")
                cur.execute("ALTER DEFAULT PRIVILEGES FOR ROLE postgres IN SCHEMA public GRANT ALL PRIVILEGES ON SEQUENCES TO gary;")
                
                conn.commit()
                print("Database schema privileges successfully ensured.")
    except Exception as e:
        print(f"Note: Could not run automated GRANT (may require superuser or already applied): {e}")

# Call this before initializing your vector store
ensure_db_permissions()
